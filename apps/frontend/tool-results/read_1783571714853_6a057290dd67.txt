   319→   319→
   320→   320→function CreateInvoiceDialog({
   321→   321→  open, onOpenChange, onCreated,
   322→   322→}: { open: boolean; onOpenChange: (v: boolean) => void; onCreated: () => void }) {
   323→   323→  const { data: patients } = useFetch<PatientLite[]>("/api/patients");
   324→   324→  const [patientId, setPatientId] = useState("");
   325→   325→  const [type, setType] = useState("consultation");
   326→   326→  const [items, setItems] = useState<InvoiceItem[]>([{ description: "", qty: 1, rate: 0, amount: 0 }]);
   327→   327→  const [discount, setDiscount] = useState(0);
   328→   328→  const [taxRate, setTaxRate] = useState(13);
   329→   329→  const [paymentMethod, setPaymentMethod] = useState("Cash");
   330→   330→  const [paid, setPaid] = useState(0);
   331→   331→  const [saving, setSaving] = useState(false);
   332→   332→
   333→   333→  const subtotal = useMemo(
   334→   334→    () => items.reduce((s, it) => s + (Number(it.qty) || 0) * (Number(it.rate) || 0), 0),
   335→   335→    [items],
   336→   336→  );
   337→   337→  const taxAmount = useMemo(() => (subtotal - discount) * (taxRate / 100), [subtotal, discount, taxRate]);
   338→   338→  const total = Math.max(0, subtotal - discount + taxAmount);
   339→   339→  const due = Math.max(0, total - paid);
   340→   340→
   341→   341→  const updateItem = (idx: number, patch: Partial<InvoiceItem>) => {
   342→   342→    setItems((prev) => prev.map((it, i) => {
   343→   343→      if (i !== idx) return it;
   344→   344→      const next = { ...it, ...patch };
   345→   345→      next.amount = (Number(next.qty) || 0) * (Number(next.rate) || 0);
   346→   346→      return next;
   347→   347→    }));
   348→   348→  };
   349→   349→
   350→   350→  const addItem = () => setItems((p) => [...p, { description: "", qty: 1, rate: 0, amount: 0 }]);
   351→   351→  const removeItem = (idx: number) => setItems((p) => p.filter((_, i) => i !== idx));
   352→   352→
   353→   353→  const reset = () => {
   354→   354→    setPatientId("");
   355→   355→    setType("consultation");
   356→   356→    setItems([{ description: "", qty: 1, rate: 0, amount: 0 }]);
   357→   357→    setDiscount(0);
   358→   358→    setTaxRate(13);
   359→   359→    setPaymentMethod("Cash");
   360→   360→    setPaid(0);
   361→   361→  };
   362→   362→
   363→   363→  const submit = async (e: React.FormEvent) => {
   364→   364→    e.preventDefault();
   365→   365→    if (!patientId) { toast.error("Please select a patient"); return; }
   366→   366→    if (items.length === 0 || items.every((i) => !i.description)) {
   367→   367→      toast.error("Add at least one invoice item"); return;
   368→   368→    }
   369→   369→    setSaving(true);
   370→   370→    try {
   371→   371→      const body = {
   372→   372→        patientId,
   373→   373→        type,
   374→   374→        subtotal,
   375→   375→        discount: Number(discount) || 0,
   376→   376→        tax: taxAmount,
   377→   377→        total,
   378→   378→        paid: Number(paid) || 0,
   379→   379→        due,
   380→   380→        status: paid >= total ? "paid" : paid > 0 ? "partial" : "unpaid",
   381→   381→        paymentMethod,
   382→   382→        items: items.filter((i) => i.description).map((i) => ({
   383→   383→          description: i.description,
   384→   384→          qty: Number(i.qty) || 0,
   385→   385→          rate: Number(i.rate) || 0,
   386→   386→          amount: (Number(i.qty) || 0) * (Number(i.rate) || 0),
   387→   387→        })),
   388→   388→      };
   389→   389→      const res = await fetch("/api/invoices", {
   390→   390→        method: "POST",
   391→   391→        headers: { "Content-Type": "application/json" },
   392→   392→        body: JSON.stringify(body),
   393→   393→      });
   394→   394→      if (!res.ok) throw new Error("Failed to create invoice");
   395→   395→      reset();
   396→   396→      onCreated();
   397→   397→    } catch {
   398→   398→      toast.error("Failed to create invoice");
   399→   399→    } finally {
   400→   400→      setSaving(false);
   401→   401→    }
   402→   402→  };
   403→   403→
   404→   404→  return (
   405→   405→    <Dialog open={open} onOpenChange={onOpenChange}>
   406→   406→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin">
   407→   407→        <DialogHeader>
   408→   408→          <DialogTitle>Create New Invoice</DialogTitle>
   409→   409→          <DialogDescription>Fill in patient and line items. Totals compute automatically.</DialogDescription>
   410→   410→        </DialogHeader>
   411→   411→        <form onSubmit={submit} className="space-y-4">
   412→   412→          {/* Patient + type */}
   413→   413→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   414→   414→            <div className="space-y-1.5">
   415→   415→              <Label>Patient *</Label>
   416→   416→              <Select value={patientId} onValueChange={setPatientId}>
   417→   417→                <SelectTrigger><SelectValue placeholder="Select patient" /></SelectTrigger>
   418→   418→                <SelectContent>
   419→   419→                  {patients?.slice(0, 100).map((p) => (
   420→   420→                    <SelectItem key={p.id} value={p.id}>
   421→   421→                      {p.name} · <span className="font-mono text-xs">{p.patientCode}</span>
   422→   422→                    </SelectItem>
   423→   423→                  ))}
   424→   424→                </SelectContent>
   425→   425→              </Select>
   426→   426→            </div>
   427→   427→            <div className="space-y-1.5">
   428→   428→              <Label>Invoice Type</Label>
   429→   429→              <Select value={type} onValueChange={setType}>
   430→   430→                <SelectTrigger><SelectValue /></SelectTrigger>
   431→   431→                <SelectContent>

   537→   537→/* ---------- Collect Payment Dialog ---------- */
   538→   538→
   539→   539→function CollectPaymentDialog({
   540→   540→  invoice, onOpenChange, onPaid,
   541→   541→}: {
   542→   542→  invoice: Invoice | null;
   543→   543→  onOpenChange: (v: boolean) => void;
   544→   544→  onPaid: () => void;
   545→   545→}) {
   546→   546→  const [amount, setAmount] = useState(0);
   547→   547→  const [method, setMethod] = useState("Cash");
   548→   548→  const [saving, setSaving] = useState(false);
   549→   549→
   550→   550→  // Reset amount when invoice changes
   551→   551→  useEffect(() => {
   552→   552→    if (invoice) {
   553→   553→      setAmount(invoice.due || 0);
   554→   554→      setMethod(invoice.paymentMethod || "Cash");
   555→   555→    }
   556→   556→  }, [invoice]);
   557→   557→
   558→   558→  if (!invoice) return null;
   559→   559→
   560→   560→  const newPaidTotal = (invoice.paid || 0) + (Number(amount) || 0);
   561→   561→  const newDue = Math.max(0, invoice.total - newPaidTotal);
   562→   562→  const newStatus = newDue <= 0 ? "paid" : newPaidTotal > 0 ? "partial" : "unpaid";
   563→   563→
   564→   564→  const submit = async (e: React.FormEvent) => {
   565→   565→    e.preventDefault();
   566→   566→    if (amount <= 0) { toast.error("Enter a valid amount"); return; }
   567→   567→    setSaving(true);
   568→   568→    try {
   569→   569→      const res = await fetch(`/api/invoices/${invoice.id}`, {
   570→   570→        method: "PATCH",
   571→   571→        headers: { "Content-Type": "application/json" },
   572→   572→        body: JSON.stringify({ paid: newPaidTotal, status: newStatus, paymentMethod: method }),
   573→   573→      });
   574→   574→      if (!res.ok) throw new Error("Failed to record payment");
   575→   575→      toast.success(`Payment of ${formatRs(amount)} recorded for ${invoice.invoiceNo}`);
   576→   576→      onPaid();
   577→   577→    } catch {
   578→   578→      toast.error("Failed to record payment");
   579→   579→    } finally {
   580→   580→      setSaving(false);
   581→   581→    }
   582→   582→  };
   583→   583→
   584→   584→  return (
   585→   585→    <Dialog open={!!invoice} onOpenChange={onOpenChange}>
   586→   586→      <DialogContent className="max-w-md">
   587→   587→        <DialogHeader>
   588→   588→          <DialogTitle className="flex items-center gap-2">
   589→   589→            <CreditCard className="w-4 h-4 text-teal-600" /> Collect Payment
   590→   590→          </DialogTitle>
   591→   591→          <DialogDescription>
   592→   592→            Invoice <span className="font-mono">{invoice.invoiceNo}</span> · {invoice.patient.name}
   593→   593→          </DialogDescription>
   594→   594→        </DialogHeader>
   595→   595→        <form onSubmit={submit} className="space-y-3">
   596→   596→          <div className="rounded-lg border bg-muted/30 p-3 text-sm space-y-1">
   597→   597→            <div className="flex justify-between"><span className="text-muted-foreground">Invoice Total</span><span className="font-medium">{formatRs(invoice.total)}</span></div>
   598→   598→            <div className="flex justify-between"><span className="text-muted-foreground">Already Paid</span><span className="text-emerald-600">{formatRs(invoice.paid)}</span></div>
   599→   599→            <div className="flex justify-between font-semibold"><span>Outstanding Due</span><span className="text-rose-600">{formatRs(invoice.due)}</span></div>
   600→   600→          </div>
   601→   601→          <div className="grid grid-cols-2 gap-3">
   602→   602→            <div className="space-y-1.5">
   603→   603→              <Label>Amount (Rs) *</Label>
   604→   604→              <Input type="number" value={amount} onChange={(e) => setAmount(Number(e.target.value))} min={0} max={invoice.due} />
   605→   605→            </div>
   606→   606→            <div className="space-y-1.5">
   607→   607→              <Label>Method</Label>
   608→   608→              <Select value={method} onValueChange={setMethod}>
   609→   609→                <SelectTrigger><SelectValue /></SelectTrigger>
   610→   610→                <SelectContent>
   611→   611→                  {PAYMENT_METHODS.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   612→   612→                </SelectContent>
   613→   613→              </Select>
   614→   614→            </div>
   615→   615→          </div>
   616→   616→          <div className="rounded-lg border bg-teal-50 dark:bg-teal-950/20 p-3 text-sm space-y-1">
   617→   617→            <div className="flex justify-between"><span className="text-muted-foreground">New Paid Total</span><span className="text-emerald-600 font-medium">{formatRs(newPaidTotal)}</span></div>
   618→   618→            <div className="flex justify-between"><span className="text-muted-foreground">Remaining Due</span><span className="font-medium">{formatRs(newDue)}</span></div>
   619→   619→            <div className="flex justify-between"><span className="text-muted-foreground">New Status</span>
   620→   620→              <Badge className={`text-[10px] ${statusColors[newStatus]}`}>{statusLabel(newStatus)}</Badge>
   621→   621→            </div>
   622→   622→          </div>
   623→   623→          <DialogFooter>
   624→   624→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   625→   625→            <Button type="submit" disabled={saving} className="bg-teal-600 hover:bg-teal-700 text-white">
   626→   626→              {saving ? "Recording…" : "Record Payment"}
   627→   627→            </Button>
   628→   628→          </DialogFooter>
   629→   629→        </form>
   630→   630→      </DialogContent>
   631→   631→    </Dialog>
   632→   632→  );
