   124→// Critical-value heuristic: highlight if the result is non-numeric OR
   125→// numeric but outside the parseable reference range. If the reference
   126→// range can't be parsed (or is missing), show normally.
   127→function isCriticalResult(result: string | null, refRange: string | null): boolean {
   128→  if (!result || !result.trim()) return false;
   129→  const r = result.trim();
   130→  const num = Number(r);
   131→  if (Number.isNaN(num)) {
   132→    // Non-numeric result — only flag critical when we have a numeric ref range
   133→    if (!refRange || !/-?\d+(\.\d+)?/.test(refRange)) return false;
   134→    return true;
   135→  }
   136→  if (!refRange) return false;
   137→  const nums = refRange.match(/-?\d+(\.\d+)?/g);
   138→  if (!nums || nums.length < 2) return false;
   139→  const [low, high] = nums.map(Number).sort((a, b) => a - b);
   140→  return num < low || num > high;
   141→}
   142→
   143→function sortList<T>(items: T[], key: keyof T | "orderedAt", dir: "asc" | "desc"): T[] {
   144→  const d = dir === "asc" ? 1 : -1;
   145→  return [...items].sort((a, b) => {
   146→    const av = a[key as keyof T];
   147→    const bv = b[key as keyof T];
   148→    if (av == null) return 1;
   149→    if (bv == null) return -1;
   150→    if (typeof av === "number" && typeof bv === "number") return (av - bv) * d;
   151→    if (key === "orderedAt") {
   152→      return (new Date(av as unknown as string).getTime() - new Date(bv as unknown as string).getTime()) * d;
   153→    }
   154→    return String(av).localeCompare(String(bv)) * d;
   155→  });
   156→}

   689→/* ---------- Lab Workflow Actions ---------- */
   690→
   691→function LabWorkflowActions({
   692→  test, onResult, onActioned,
   693→}: {
   694→  test: LabTest;
   695→  onResult: () => void;
   696→  onActioned: () => void;
   697→}) {
   698→  const [saving, setSaving] = useState(false);
   699→
   700→  const patch = async (body: Record<string, unknown>, successMsg: string) => {
   701→    setSaving(true);
   702→    try {
   703→      const res = await fetch(`/api/lab-tests/${test.id}`, {
   704→        method: "PATCH",
   705→        headers: { "Content-Type": "application/json" },
   706→        body: JSON.stringify(body),
   707→      });
   708→      if (!res.ok) throw new Error("Failed");
   709→      toast.success(successMsg);
   710→      onActioned();
   711→    } catch {
   712→      toast.error("Action failed");
   713→    } finally {
   714→      setSaving(false);
   715→    }
   716→  };
   717→
   718→  if (test.status === "pending") {
   719→    return (
   720→      <Button variant="outline" size="sm" className="h-7 gap-1 text-xs" disabled={saving}
   721→        onClick={() => patch({ status: "collected" }, `Sample collected for ${test.testCode}`)}>
   722→        <Syringe className="w-3 h-3" /> Collect
   723→      </Button>
   724→    );
   725→  }
   726→  if (test.status === "collected") {
   727→    return (
   728→      <Button variant="outline" size="sm" className="h-7 gap-1 text-xs" disabled={saving}
   729→        onClick={() => patch({ status: "processing" }, `${test.testName} processing started`)}>
   730→        <Microscope className="w-3 h-3" /> Process
   731→      </Button>
   732→    );
   733→  }
   734→  if (test.status === "processing") {
   735→    return (
   736→      <Button variant="outline" size="sm" className="h-7 gap-1 text-xs bg-amber-50 hover:bg-amber-100 border-amber-200 dark:bg-amber-950/30 dark:hover:bg-amber-950/50 dark:border-amber-900"
   737→        onClick={onResult}>
   738→        <FileText className="w-3 h-3" /> Enter Result
   739→      </Button>
   740→    );
   741→  }
   742→  if (test.status === "completed") {
   743→    return (
   744→      <Button size="sm" className="h-7 gap-1 text-xs bg-emerald-600 hover:bg-emerald-700 text-white" disabled={saving}
   745→        onClick={() => patch({ status: "approved" }, `${test.testCode} approved`)}>
   746→        <CheckCheck className="w-3 h-3" /> Approve
   747→      </Button>
   748→    );
   749→  }
   750→  return (
   751→    <Badge variant="outline" className="text-[10px] text-emerald-600 border-emerald-200 dark:border-emerald-900">
   752→      <CheckCheck className="w-3 h-3 mr-1" /> Done
   753→    </Badge>
   754→  );
   755→}

   968→          unit: unit.trim() || null,
