    26→

    89→

    97→

   100→  const { data, loading } = useFetch<LabDashboardData>(
   101→    refresh ? `/api/lab-dashboard?_r=${refresh}` : "/api/lab-dashboard",
   102→  );

   118→    ];
   119→  }, [data]);
   120→
   121→  if (loading || !data) return <DashboardSkeleton />;
   122→
   123→  const { kpis } = data;
   124→
   125→  /* ---------- 10 KPI cards ---------- */

   129→    { key: "collectedSamples", label: "Collected Samples", value: String(kpis.collectedSamples), icon: Syringe, accent: "from-cyan-500 to-teal-600" },
   130→    { key: "processingSamples", label: "Processing", value: String(kpis.processingSamples), icon: Microscope, accent: "from-violet-500 to-purple-600" },
   131→    { key: "pendingResults", label: "Pending Results", value: String(kpis.pendingResults), icon: TestTube, accent: "from-amber-500 to-yellow-600" },
   132→    { key: "pendingApproval", label: "Pending Approval", value: String(kpis.pendingApproval), icon: FileText, accent: "from-teal-500 to-emerald-600" },
   133→    { key: "completedReports", label: "Completed Reports", value: String(kpis.completedReports), icon: CheckCircle2, accent: "from-emerald-500 to-emerald-600" },
   134→    { key: "criticalResults", label: "Critical Results", value: String(kpis.criticalResults), icon: AlertTriangle, accent: "from-rose-500 to-rose-600" },
   135→    { key: "todayRevenue", label: "Today's Revenue", value: formatRs(kpis.todayRevenue), icon: Wallet, accent: "from-emerald-500 to-teal-600" },
   136→    { key: "avgTAT", label: "Avg TAT (hrs)", value: String(kpis.avgTAT), icon: Clock, accent: "from-teal-500 to-cyan-600" },
   137→  ];
   138→
   139→  const handleExport = () => {

   142→      ["Pending Collection", kpis.pendingCollection],
   143→      ["Collected Samples", kpis.collectedSamples],
   144→      ["Processing Samples", kpis.processingSamples],
   145→      ["Pending Results", kpis.pendingResults],
   146→      ["Pending Approval", kpis.pendingApproval],
   147→      ["Completed Reports", kpis.completedReports],
   148→      ["Critical Results", kpis.criticalResults],
   149→      ["Today's Revenue", formatRs(kpis.todayRevenue)],
   150→      ["Avg TAT (hours)", kpis.avgTAT],
   151→      ["Month Revenue", formatRs(data.monthRevenue)],
   152→      ["QC Total", data.qcStats.total],
   153→      ["QC Pass", data.qcStats.pass],
   154→      ["QC Fail", data.qcStats.fail],
   155→      ["QC Warning", data.qcStats.warning],

   159→  };
   160→
   161→  const handleRefresh = () => setRefresh((r) => r + 1);
   162→
   163→  return (

   165→      {/* Header */}
   166→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   167→        <div>
   168→          <h2 className="text-xl font-bold flex items-center gap-2">
   169→            <FlaskConical className="w-5 h-5 text-teal-600" /> Laboratory Dashboard
   170→          </h2>
   171→          <p className="text-sm text-muted-foreground">
   172→            Real-time LIMS analytics · Month revenue {formatRs(data.monthRevenue)}
   173→          </p>
   174→        </div>
   175→        <div className="flex items-center gap-2">

   177→            <Activity className="w-4 h-4" /> Refresh
   178→          </Button>
   179→          <Button variant="outline" size="sm" className="gap-1.5" onClick={handleExport}>
   180→            <Download className="w-4 h-4" /> Export CSV

   182→        </div>

   184→
   185→      {/* 10 KPI cards */}

   193→          >
   194→            <Card className="overflow-hidden border-border/60">

   198→                    <k.icon className="w-4.5 h-4.5" />
   199→                  </div>
   200→                </div>
   201→                <p className="text-xl font-bold mt-2 leading-tight">{k.value}</p>
   202→                <p className="text-[11px] text-muted-foreground leading-tight mt-0.5">{k.label}</p>
   203→              </CardContent>
   204→            </Card>
   205→          </motion.div>
   206→        ))}
   207→      </div>
   208→
   209→      {/* Charts row 1 — Daily volume (2-col) + Dept requests donut */}

   211→        <Card className="lg:col-span-2">
   212→          <CardHeader className="pb-2">
   213→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   214→              <Activity className="w-4 h-4 text-teal-600" /> Daily Test Volume (7 days)

   221→                  <defs>

   237→
   238→        <Card>
   239→          <CardHeader className="pb-2">
   240→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   241→              <Building2 className="w-4 h-4 text-teal-600" /> Department-wise Requests
   242→            </CardTitle>
   243→          </CardHeader>
   244→          <CardContent>
   245→            <div style={{ height: 250 }}>
   246→              {data.deptRequests.length === 0 ? (
   247→                <div className="h-full flex items-center justify-center text-xs text-muted-foreground">
   248→                  No department requests
   249→                </div>
   250→              ) : (
   251→                <ResponsiveContainer width="100%" height="100%">
   252→                  <PieChart>
   253→                    <Pie
   254→                      data={data.deptRequests}
   255→                      dataKey="value"
   256→                      nameKey="name"
   257→                      cx="50%"
   258→                      cy="50%"
   259→                      innerRadius={55}
   260→                      outerRadius={85}
   261→                      paddingAngle={2}
   262→                    >
   263→                      {data.deptRequests.map((d, i) => (
   264→                        <Cell key={i} fill={d.color || "#0d9488"} />
   265→                      ))}
   266→                    </Pie>
   267→                    <Tooltip content={<ChartTooltip />} />
   268→                    <Legend
   269→                      layout="horizontal"
   270→                      verticalAlign="bottom"
   271→                      align="center"
   272→                      iconType="circle"
   273→                      wrapperStyle={{ fontSize: 10 }}
   274→                    />
   275→                  </PieChart>
   276→                </ResponsiveContainer>
   277→              )}
   278→            </div>
   279→          </CardContent>
   280→        </Card>
   281→      </div>
   282→
   283→      {/* Charts row 2 — Most Requested (horizontal bar) + Tech Performance */}
   284→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">

   289→            </CardTitle>
   290→          </CardHeader>
   291→          <CardContent>
   292→            <div style={{ height: 280 }}>
   293→              {data.mostRequested.length === 0 ? (
   294→                <div className="h-full flex items-center justify-center text-xs text-muted-foreground">
   295→                  No test request data
   296→                </div>
   297→              ) : (
   298→                <ResponsiveContainer width="100%" height="100%">
   299→                  <BarChart
   300→                    data={data.mostRequested}
   301→                    layout="vertical"
   302→                    margin={{ left: 16, right: 16, top: 0, bottom: 0 }}

   306→                    <YAxis
   307→                      type="category"
   308→                      dataKey="name"
   309→                      tick={{ fontSize: 10 }}
   310→                      className="fill-muted-foreground"
   311→                      width={120}
   312→                      tickFormatter={(v: string) => (v.length > 18 ? v.slice(0, 18) + "…" : v)}
   313→                    />
   314→                    <Tooltip content={<ChartTooltip />} />
   315→                    <Bar dataKey="count" fill="#0d9488" radius={[0, 4, 4, 0]} barSize={18} />
   316→                  </BarChart>
   317→                </ResponsiveContainer>
   318→              )}
   319→            </div>
   320→          </CardContent>
   321→        </Card>
   322→
   323→        <Card>
   324→          <CardHeader className="pb-2">
   325→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   326→              <Stethoscope className="w-4 h-4 text-emerald-600" /> Technician Performance
   327→            </CardTitle>
   328→          </CardHeader>
   329→          <CardContent>
   330→            <div style={{ height: 280 }}>
   331→              {data.techPerf.length === 0 ? (
   332→                <div className="h-full flex items-center justify-center text-xs text-muted-foreground">
   333→                  No technician data
   334→                </div>
   335→              ) : (
   336→                <ResponsiveContainer width="100%" height="100%">
   337→                  <BarChart data={data.techPerf} margin={{ left: -16, right: 8, top: 8, bottom: 0 }}>
   338→                    <CartesianGrid strokeDasharray="3 3" stroke="#e2e8f0" vertical={false} />

   340→                      dataKey="name"
   341→                      tick={{ fontSize: 10 }}
   342→                      className="fill-muted-foreground"
   343→                      tickFormatter={(v: string) => (v.length > 10 ? v.split(" ")[0] : v)}
   344→                    />
   345→                    <YAxis className="fill-muted-foreground" tick={{ fontSize: 11 }} allowDecimals={false} />
   346→                    <Tooltip content={<ChartTooltip />} />
   347→                    <Bar dataKey="completed" fill="#10b981" radius={[4, 4, 0, 0]} barSize={28} />
   348→                  </BarChart>
   349→                </ResponsiveContainer>
   350→              )}
   351→            </div>
   352→          </CardContent>
   353→        </Card>
   354→      </div>
   355→
   356→      {/* Charts row 3 — Abnormal Result Stats (2-col) + QC Stats card */}
   357→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4">
   358→        <Card className="lg:col-span-2">
   359→          <CardHeader className="pb-2">
   360→            <CardTitle className="text-sm font-medium flex items-center gap-2">
   361→              <AlertTriangle className="w-4 h-4 text-rose-600" /> Abnormal Result Statistics

   390→          <CardHeader className="pb-2">

   392→              <CheckCircle2 className="w-4 h-4 text-emerald-600" /> QC Statistics

   395→          <CardContent className="space-y-3">

   400→              </div>

   402→                <p className="text-2xl font-bold text-rose-700 dark:text-rose-400">{data.qcStats.fail}</p>

   445→      {/* Live panels — 2-col grid */}
   446→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">

   448→        <Card>

   454→          </CardHeader>
   455→          <CardContent className="pt-0">

   460→                data.waitingCollection.map((w) => (

   692→          </div>

   695→    </div>

   697→}
