    98→/* ---------- Invoice print HTML builder ---------- */
    99→function buildInvoiceHTML(inv: Invoice): string {
   100→  const statusClass =
   101→    inv.status === "paid" ? "emerald" :
   102→    inv.status === "partial" ? "teal" :
   103→    inv.status === "refunded" ? "rose" : "rose";
   104→  const statusBadge = `<span class="badge ${statusClass}">${statusLabel(inv.status)}</span>`;
   105→
   106→  const patientGrid = `
   107→    <div class="info-grid">
   108→      <div><div class="label">Bill To</div><div><strong>${escapeHTML(inv.patient.name)}</strong></div>
   109→        <div>Code: <span style="font-family:monospace">${escapeHTML(inv.patient.patientCode)}</span></div>
   110→        <div>Phone: ${escapeHTML(inv.patient.phone || "—")}</div></div>
   111→      <div><div class="label">Type</div><div>${escapeHTML(inv.type)}</div>
   112→        <div class="label" style="margin-top:6px">Payment Method</div>
   113→        <div>${escapeHTML(inv.paymentMethod || "—")}</div></div>
   114→    </div>`;
   115→
   116→  const itemRows = (inv.items?.length ? inv.items : []).map((it) => `
   117→    <tr>
   118→      <td>${escapeHTML(it.description)}</td>
   119→      <td style="text-align:right">${it.qty}</td>
   120→      <td style="text-align:right">${formatRs(it.rate)}</td>
   121→      <td style="text-align:right">${formatRs(it.amount)}</td>
   122→    </tr>`).join("");
   123→
   124→  const itemsTable = `
   125→    <h2>Items</h2>
   126→    <table>
   127→      <thead><tr><th>Description</th><th style="text-align:right">Qty</th><th style="text-align:right">Rate</th><th style="text-align:right">Amount</th></tr></thead>
   128→      <tbody>${itemRows || `<tr><td colspan="4" style="text-align:center;color:#94a3b8">No items</td></tr>`}</tbody>
   129→    </table>`;
   130→
   131→  const totals = `
   132→    <div class="totals">
   133→      <div class="row"><span>Subtotal</span><span>${formatRs(inv.subtotal)}</span></div>
   134→      <div class="row"><span>Discount</span><span>- ${formatRs(inv.discount)}</span></div>
   135→      <div class="row"><span>Tax</span><span>+ ${formatRs(inv.tax)}</span></div>
   136→      <div class="row grand"><span>Total</span><span>${formatRs(inv.total)}</span></div>
   137→      <div class="row"><span>Paid</span><span>${formatRs(inv.paid)}</span></div>
   138→      <div class="row"><span>Due</span><span>${formatRs(inv.due)}</span></div>
   139→    </div>`;
   140→
   141→  return `${docHeader(inv.invoiceNo, "INVOICE", formatDate(inv.date), statusBadge)}
   142→    ${patientGrid}
   143→    ${itemsTable}
   144→    ${totals}
   145→    <div class="signature">
   146→      <div class="sig-block"><div class="line"></div><div class="name">Received By</div><div class="role">Patient / Guardian</div></div>
   147→      <div class="sig-block"><div class="line"></div><div class="name">Authorized Signatory</div><div class="role">MedCore Health Center</div></div>
   148→    </div>`;
   149→}
   150→
   151→function escapeHTML(s: string): string {
   152→  return String(s ?? "").replace(/[&<>"']/g, (c) =>
   153→    ({ "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c] as string));
   154→}
   155→
   156→function printInvoice(inv: Invoice) {
   157→  printHTML(`Invoice ${inv.invoiceNo}`, buildInvoiceHTML(inv));
   158→}

   160→export function BillingView() {
   161→  const [refreshKey, setRefreshKey] = useState(0);
   162→  const refresh = useCallback(() => setRefreshKey((k) => k + 1), []);
   163→  const { data: invoices, loading, error } = useFetch<Invoice[]>(
   164→    refreshKey ? `/api/invoices?_r=${refreshKey}` : "/api/invoices",
   165→  );
   166→  const [q, setQ] = useState("");
   167→  const [statusFilter, setStatusFilter] = useState<string>("all");
   168→  const [sortKey, setSortKey] = useState<SortKey>("date");
   169→  const [sortDir, setSortDir] = useState<"asc" | "desc">("desc");
   170→  const [createOpen, setCreateOpen] = useState(false);
   171→  const [viewId, setViewId] = useState<string | null>(null);
   172→  const [payInvoice, setPayInvoice] = useState<Invoice | null>(null);
   173→  const [refundInvoice, setRefundInvoice] = useState<Invoice | null>(null);
   174→
   175→  const filtered = useMemo(() => {
   176→    if (!invoices) return [];
   177→    const ql = q.toLowerCase();
   178→    const list = invoices.filter((inv) => {
   179→      const matchesStatus = statusFilter === "all" || inv.status === statusFilter;
   180→      const matchesSearch = !ql ||
   181→        inv.invoiceNo.toLowerCase().includes(ql) ||
   182→        inv.patient.name.toLowerCase().includes(ql);
   183→      return matchesStatus && matchesSearch;
   184→    });
   185→    const dir = sortDir === "asc" ? 1 : -1;
   186→    return list.sort((a, b) => {
   187→      let av: number | string;
   188→      let bv: number | string;
   189→      if (sortKey === "date") { av = new Date(a.date).getTime(); bv = new Date(b.date).getTime(); }
   190→      else if (sortKey === "invoiceNo") { av = a.invoiceNo; bv = b.invoiceNo; }
   191→      else { av = (a as unknown as Record<string, number>)[sortKey]; bv = (b as unknown as Record<string, number>)[sortKey]; }
   192→      if (typeof av === "number" && typeof bv === "number") return (av - bv) * dir;
   193→      return String(av).localeCompare(String(bv)) * dir;
   194→    });
   195→  }, [invoices, q, statusFilter, sortKey, sortDir]);
   196→
   197→  const { paged, page, totalPages, setPage, size, setSize, range } = usePagination<Invoice>(filtered, 10);
   198→
   199→  const stats = useMemo(() => {
   200→    if (!invoices) return { revenue: 0, collected: 0, due: 0, overdue: 0 };
   201→    return {
   202→      revenue: invoices.reduce((s, i) => s + (i.total || 0), 0),
   203→      collected: invoices.reduce((s, i) => s + (i.paid || 0), 0),
   204→      due: invoices.reduce((s, i) => s + (i.due || 0), 0),
   205→      overdue: invoices.filter((i) => i.status === "unpaid").length,
   206→    };
   207→  }, [invoices]);
   208→
   209→  const selected = invoices?.find((i) => i.id === viewId) || null;
   210→
   211→  const handleExport = () => {
   212→    if (!filtered.length) { toast.info("No invoices to export"); return; }
   213→    exportToCSV("invoices", [
   214→      "Invoice No", "Date", "Patient", "Type", "Subtotal", "Discount", "Tax",
   215→      "Total", "Paid", "Due", "Status", "Method",
   216→    ], filtered.map((i) => [
   217→      i.invoiceNo, formatDate(i.date), i.patient.name, i.type,
   218→      i.subtotal, i.discount, i.tax, i.total, i.paid, i.due,
   219→      statusLabel(i.status), i.paymentMethod || "",
   220→    ]));
   221→    toast.success(`Exported ${filtered.length} invoices to CSV`);
   222→  };
   223→
   224→  if (error) {
   225→    return (
   226→      <div className="space-y-4 animate-fade-in">
   227→        <Card>
   228→          <CardContent className="p-10 text-center text-sm text-rose-600">
   229→            Failed to load invoices: {error}
   230→          </CardContent>
   231→        </Card>
   232→      </div>
   233→    );
   234→  }
   235→
   236→  return (
   237→    <div className="space-y-4 animate-fade-in">
   238→      {/* Header */}
   239→      <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3">
   240→        <div>
   241→          <h2 className="text-xl font-bold flex items-center gap-2">
   242→            <Receipt className="w-5 h-5 text-teal-600" /> Billing &amp; Invoices
   243→          </h2>
   244→          <p className="text-sm text-muted-foreground">
   245→            {invoices?.length ?? 0} invoices · {formatRs(stats.collected)} collected
   246→          </p>
   247→        </div>
   248→        <div className="flex items-center gap-2">
   249→          <Button variant="outline" size="sm" className="gap-1.5" onClick={handleExport}>
   250→            <Download className="w-4 h-4" /> Export CSV
   251→          </Button>
   252→          <Button
   253→            size="sm"
   254→            className="gap-1.5 bg-teal-600 hover:bg-teal-700 text-white"
   255→            onClick={() => setCreateOpen(true)}

   257→            <Plus className="w-4 h-4" /> Create Invoice
   258→          </Button>
   259→        </div>
   260→      </div>
   261→
   262→      {/* Stat cards */}
   263→      <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 sm:gap-4">
   264→        {[
   265→          { label: "Total Revenue", value: formatRs(stats.revenue), icon: TrendingUp, accent: "from-teal-500 to-teal-600" },
   266→          { label: "Collected", value: formatRs(stats.collected), icon: Wallet, accent: "from-emerald-500 to-emerald-600" },
   267→          { label: "Outstanding Due", value: formatRs(stats.due), icon: AlertTriangle, accent: "from-amber-500 to-orange-500" },
   268→          { label: "Overdue Invoices", value: String(stats.overdue), icon: Receipt, accent: "from-rose-500 to-rose-600" },

   274→            transition={{ delay: i * 0.05 }}
   275→          >
   276→            <Card className="hover:shadow-md transition-shadow">
   277→              <CardContent className="p-4 sm:p-5">
   278→                <div className="flex items-start justify-between">
   279→                  <div className={`w-10 h-10 rounded-xl bg-gradient-to-br ${s.accent} flex items-center justify-center shadow-sm`}>
   280→                    <s.icon className="w-5 h-5 text-white" />
   281→                  </div>
   282→                </div>
   283→                <p className="mt-3 text-xl sm:text-2xl font-bold tracking-tight">{s.value}</p>
   284→                <p className="text-xs sm:text-sm text-muted-foreground mt-0.5">{s.label}</p>

   493→function CreateInvoiceDialog({
   494→  open, onOpenChange, onCreated,
   495→}: { open: boolean; onOpenChange: (v: boolean) => void; onCreated: () => void }) {
   496→  const { data: patients } = useFetch<PatientLite[]>("/api/patients");
   497→  const [patientId, setPatientId] = useState("");
   498→  const [type, setType] = useState("consultation");
   499→  const [items, setItems] = useState<InvoiceItem[]>([{ description: "", qty: 1, rate: 0, amount: 0 }]);
   500→  const [discount, setDiscount] = useState(0);
   501→  const [taxRate, setTaxRate] = useState(13);
   502→  const [paymentMethod, setPaymentMethod] = useState("Cash");
   503→  const [paid, setPaid] = useState(0);
   504→  const [saving, setSaving] = useState(false);
   505→
   506→  const subtotal = useMemo(
   507→    () => items.reduce((s, it) => s + (Number(it.qty) || 0) * (Number(it.rate) || 0), 0),
   508→    [items],
   509→  );
   510→  const taxAmount = useMemo(() => (subtotal - discount) * (taxRate / 100), [subtotal, discount, taxRate]);
   511→  const total = Math.max(0, subtotal - discount + taxAmount);
   512→  const due = Math.max(0, total - paid);
   513→
   514→  const updateItem = (idx: number, patch: Partial<InvoiceItem>) => {
   515→    setItems((prev) => prev.map((it, i) => {
   516→      if (i !== idx) return it;
   517→      const next = { ...it, ...patch };
   518→      next.amount = (Number(next.qty) || 0) * (Number(next.rate) || 0);
   519→      return next;
   520→    }));
   521→  };
   522→
   523→  const addItem = () => setItems((p) => [...p, { description: "", qty: 1, rate: 0, amount: 0 }]);
   524→  const removeItem = (idx: number) => setItems((p) => p.filter((_, i) => i !== idx));
   525→
   526→  const reset = () => {
   527→    setPatientId("");
   528→    setType("consultation");
   529→    setItems([{ description: "", qty: 1, rate: 0, amount: 0 }]);
   530→    setDiscount(0);
   531→    setTaxRate(13);
   532→    setPaymentMethod("Cash");
   533→    setPaid(0);
   534→  };
   535→
   536→  const submit = async (e: React.FormEvent) => {
   537→    e.preventDefault();

   539→    if (items.length === 0 || items.every((i) => !i.description)) {
   540→      toast.error("Add at least one invoice item"); return;
   541→    }
   542→    setSaving(true);
   543→    try {
   544→      const body = {
   545→        patientId,
   546→        type,
   547→        subtotal,
   548→        discount: Number(discount) || 0,
   549→        tax: taxAmount,
   550→        total,
   551→        paid: Number(paid) || 0,
   552→        due,
   553→        status: paid >= total ? "paid" : paid > 0 ? "partial" : "unpaid",
   554→        paymentMethod,
   555→        items: items.filter((i) => i.description).map((i) => ({
   556→          description: i.description,
   557→          qty: Number(i.qty) || 0,
   558→          rate: Number(i.rate) || 0,
   559→          amount: (Number(i.qty) || 0) * (Number(i.rate) || 0),
   560→        })),
   561→      };
   562→      const res = await fetch("/api/invoices", {
   563→        method: "POST",
   564→        headers: { "Content-Type": "application/json" },
   565→        body: JSON.stringify(body),
   566→      });
   567→      if (!res.ok) throw new Error("Failed to create invoice");
   568→      reset();
   569→      onCreated();
   570→    } catch {
   571→      toast.error("Failed to create invoice");
   572→    } finally {
   573→      setSaving(false);
   574→    }
   575→  };
   576→
   577→  return (

   710→/* ---------- Collect Payment Dialog ---------- */
   711→
   712→function CollectPaymentDialog({
   713→  invoice, onOpenChange, onPaid,
   714→}: {
   715→  invoice: Invoice | null;
   716→  onOpenChange: (v: boolean) => void;
   717→  onPaid: () => void;
   718→}) {
   719→  const [amount, setAmount] = useState(0);
   720→  const [method, setMethod] = useState("Cash");
   721→  const [saving, setSaving] = useState(false);
   722→
   723→  // Reset amount when invoice changes
   724→  useEffect(() => {
   725→    if (invoice) {
   726→      setAmount(invoice.due || 0);
   727→      setMethod(invoice.paymentMethod || "Cash");
   728→    }
   729→  }, [invoice]);
   730→
   731→  if (!invoice) return null;
   732→
   733→  const newPaidTotal = (invoice.paid || 0) + (Number(amount) || 0);
   734→  const newDue = Math.max(0, invoice.total - newPaidTotal);
   735→  const newStatus = newDue <= 0 ? "paid" : newPaidTotal > 0 ? "partial" : "unpaid";
   736→
   737→  const submit = async (e: React.FormEvent) => {
   738→    e.preventDefault();
   739→    if (amount <= 0) { toast.error("Enter a valid amount"); return; }
   740→    setSaving(true);
   741→    try {
   742→      const res = await fetch(`/api/invoices/${invoice.id}`, {
   743→        method: "PATCH",
   744→        headers: { "Content-Type": "application/json" },
   745→        body: JSON.stringify({ paid: newPaidTotal, status: newStatus, paymentMethod: method }),
   746→      });
   747→      if (!res.ok) throw new Error("Failed to record payment");
   748→      toast.success(`Payment of ${formatRs(amount)} recorded for ${invoice.invoiceNo}`);
   749→      onPaid();
   750→    } catch {
   751→      toast.error("Failed to record payment");
   752→    } finally {
   753→      setSaving(false);
   754→    }
   755→  };
   756→
   757→  return (
   758→    <Dialog open={!!invoice} onOpenChange={onOpenChange}>
   759→      <DialogContent className="max-w-md">
   760→        <DialogHeader>
   761→          <DialogTitle className="flex items-center gap-2">
   762→            <CreditCard className="w-4 h-4 text-teal-600" /> Collect Payment
   763→          </DialogTitle>
   764→          <DialogDescription>
   765→            Invoice <span className="font-mono">{invoice.invoiceNo}</span> · {invoice.patient.name}
   766→          </DialogDescription>
   767→        </DialogHeader>
   768→        <form onSubmit={submit} className="space-y-3">
   769→          <div className="rounded-lg border bg-muted/30 p-3 text-sm space-y-1">
   770→            <div className="flex justify-between"><span className="text-muted-foreground">Invoice Total</span><span className="font-medium">{formatRs(invoice.total)}</span></div>
   771→            <div className="flex justify-between"><span className="text-muted-foreground">Already Paid</span><span className="text-emerald-600">{formatRs(invoice.paid)}</span></div>
   772→            <div className="flex justify-between font-semibold"><span>Outstanding Due</span><span className="text-rose-600">{formatRs(invoice.due)}</span></div>
   773→          </div>
   774→          <div className="grid grid-cols-2 gap-3">
   775→            <div className="space-y-1.5">
   776→              <Label>Amount (Rs) *</Label>
   777→              <Input type="number" value={amount} onChange={(e) => setAmount(Number(e.target.value))} min={0} max={invoice.due} />
   778→            </div>
   779→            <div className="space-y-1.5">
   780→              <Label>Method</Label>
   781→              <Select value={method} onValueChange={setMethod}>
   782→                <SelectTrigger><SelectValue /></SelectTrigger>
   783→                <SelectContent>
   784→                  {PAYMENT_METHODS.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   785→                </SelectContent>
   786→              </Select>
   787→            </div>
   788→          </div>
   789→          <div className="rounded-lg border bg-teal-50 dark:bg-teal-950/20 p-3 text-sm space-y-1">
   790→            <div className="flex justify-between"><span className="text-muted-foreground">New Paid Total</span><span className="text-emerald-600 font-medium">{formatRs(newPaidTotal)}</span></div>
   791→            <div className="flex justify-between"><span className="text-muted-foreground">Remaining Due</span><span className="font-medium">{formatRs(newDue)}</span></div>
   792→            <div className="flex justify-between"><span className="text-muted-foreground">New Status</span>
   793→              <Badge className={`text-[10px] ${statusColors[newStatus]}`}>{statusLabel(newStatus)}</Badge>
   794→            </div>
   795→          </div>
   796→          <DialogFooter>
   797→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   798→            <Button type="submit" disabled={saving} className="bg-teal-600 hover:bg-teal-700 text-white">
   799→              {saving ? "Recording…" : "Record Payment"}
