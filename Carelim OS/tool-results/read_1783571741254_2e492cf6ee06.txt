   318→/* ---------- Create Invoice Dialog ---------- */
   319→
   320→function CreateInvoiceDialog({
   321→  open, onOpenChange, onCreated,
   322→}: { open: boolean; onOpenChange: (v: boolean) => void; onCreated: () => void }) {
   323→  const { data: patients } = useFetch<PatientLite[]>("/api/patients");
   324→  const [patientId, setPatientId] = useState("");
   325→  const [type, setType] = useState("consultation");
   326→  const [items, setItems] = useState<InvoiceItem[]>([{ description: "", qty: 1, rate: 0, amount: 0 }]);
   327→  const [discount, setDiscount] = useState(0);
   328→  const [taxRate, setTaxRate] = useState(13);
   329→  const [paymentMethod, setPaymentMethod] = useState("Cash");
   330→  const [paid, setPaid] = useState(0);
   331→  const [saving, setSaving] = useState(false);
   332→
   333→  const subtotal = useMemo(
   334→    () => items.reduce((s, it) => s + (Number(it.qty) || 0) * (Number(it.rate) || 0), 0),
   335→    [items],
   336→  );
   337→  const taxAmount = useMemo(() => (subtotal - discount) * (taxRate / 100), [subtotal, discount, taxRate]);
   338→  const total = Math.max(0, subtotal - discount + taxAmount);
   339→  const due = Math.max(0, total - paid);
   340→
   341→  const updateItem = (idx: number, patch: Partial<InvoiceItem>) => {
   342→    setItems((prev) => prev.map((it, i) => {
   343→      if (i !== idx) return it;
   344→      const next = { ...it, ...patch };
   345→      next.amount = (Number(next.qty) || 0) * (Number(next.rate) || 0);
   346→      return next;
   347→    }));
   348→  };
   349→
   350→  const addItem = () => setItems((p) => [...p, { description: "", qty: 1, rate: 0, amount: 0 }]);
   351→  const removeItem = (idx: number) => setItems((p) => p.filter((_, i) => i !== idx));
   352→
   353→  const reset = () => {
   354→    setPatientId("");
   355→    setType("consultation");
   356→    setItems([{ description: "", qty: 1, rate: 0, amount: 0 }]);
   357→    setDiscount(0);
   358→    setTaxRate(13);
   359→    setPaymentMethod("Cash");
   360→    setPaid(0);
   361→  };
   362→
   363→  const submit = async (e: React.FormEvent) => {
   364→    e.preventDefault();
   365→    if (!patientId) { toast.error("Please select a patient"); return; }
   366→    if (items.length === 0 || items.every((i) => !i.description)) {
   367→      toast.error("Add at least one invoice item"); return;
   368→    }
   369→    setSaving(true);
   370→    try {
   371→      const body = {
   372→        patientId,
   373→        type,
   374→        subtotal,
   375→        discount: Number(discount) || 0,
   376→        tax: taxAmount,
   377→        total,
   378→        paid: Number(paid) || 0,
   379→        due,
   380→        status: paid >= total ? "paid" : paid > 0 ? "partial" : "unpaid",
   381→        paymentMethod,
   382→        items: items.filter((i) => i.description).map((i) => ({
   383→          description: i.description,
   384→          qty: Number(i.qty) || 0,
   385→          rate: Number(i.rate) || 0,
   386→          amount: (Number(i.qty) || 0) * (Number(i.rate) || 0),
   387→        })),
   388→      };
   389→      const res = await fetch("/api/invoices", {
   390→        method: "POST",
   391→        headers: { "Content-Type": "application/json" },
   392→        body: JSON.stringify(body),
   393→      });
   394→      if (!res.ok) throw new Error("Failed to create invoice");
   395→      reset();
   396→      onCreated();
   397→    } catch {
   398→      toast.error("Failed to create invoice");
   399→    } finally {
   400→      setSaving(false);
   401→    }
   402→  };
   403→
   404→  return (
   405→    <Dialog open={open} onOpenChange={onOpenChange}>
   406→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto scrollbar-thin">
   407→        <DialogHeader>
   408→          <DialogTitle>Create New Invoice</DialogTitle>
   409→          <DialogDescription>Fill in patient and line items. Totals compute automatically.</DialogDescription>
   410→        </DialogHeader>
   411→        <form onSubmit={submit} className="space-y-4">
   412→          {/* Patient + type */}
   413→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   414→            <div className="space-y-1.5">
   415→              <Label>Patient *</Label>
   416→              <Select value={patientId} onValueChange={setPatientId}>
   417→                <SelectTrigger><SelectValue placeholder="Select patient" /></SelectTrigger>
   418→                <SelectContent>
   419→                  {patients?.slice(0, 100).map((p) => (
   420→                    <SelectItem key={p.id} value={p.id}>
   421→                      {p.name} · <span className="font-mono text-xs">{p.patientCode}</span>
   422→                    </SelectItem>
   423→                  ))}
   424→                </SelectContent>
   425→              </Select>
   426→            </div>
   427→            <div className="space-y-1.5">
   428→              <Label>Invoice Type</Label>
   429→              <Select value={type} onValueChange={setType}>
   430→                <SelectTrigger><SelectValue /></SelectTrigger>
   431→                <SelectContent>
   432→                  <SelectItem value="consultation">Consultation</SelectItem>
   433→                  <SelectItem value="pharmacy">Pharmacy</SelectItem>
   434→                  <SelectItem value="lab">Lab</SelectItem>
   435→                  <SelectItem value="package">Package</SelectItem>
   436→                </SelectContent>
   437→              </Select>
   438→            </div>
   439→          </div>
   440→
   441→          {/* Items */}
   442→          <div className="space-y-2">
   443→            <div className="flex items-center justify-between">
   444→              <Label>Line Items</Label>
   445→              <Button type="button" variant="outline" size="sm" className="h-7 gap-1 text-xs" onClick={addItem}>
   446→                <Plus className="w-3 h-3" /> Add Item
   447→              </Button>
   448→            </div>
   449→            <div className="space-y-2 rounded-lg border p-2 bg-muted/30 max-h-56 overflow-y-auto scrollbar-thin">
   450→              {items.map((it, idx) => (
   451→                <div key={idx} className="grid grid-cols-12 gap-2 items-center">
   452→                  <Input
   453→                    className="col-span-12 sm:col-span-5 h-8 text-sm"
   454→                    placeholder="Description"
   455→                    value={it.description}
   456→                    onChange={(e) => updateItem(idx, { description: e.target.value })}
   457→                  />
   458→                  <Input
   459→                    type="number"
   460→                    className="col-span-3 sm:col-span-2 h-8 text-sm"
   461→                    placeholder="Qty"
   462→                    value={it.qty}
   463→                    onChange={(e) => updateItem(idx, { qty: Number(e.target.value) })}
   464→                  />
   465→                  <Input
   466→                    type="number"
   467→                    className="col-span-4 sm:col-span-2 h-8 text-sm"
   468→                    placeholder="Rate"
   469→                    value={it.rate}
   470→                    onChange={(e) => updateItem(idx, { rate: Number(e.target.value) })}
   471→                  />
   472→                  <div className="col-span-4 sm:col-span-2 text-right text-sm font-medium">
   473→                    {formatRs(it.amount)}
   474→                  </div>
   475→                  <Button
   476→                    type="button"
   477→                    variant="ghost"
   478→                    size="sm"
   479→                    className="col-span-1 h-8 w-8 p-0 text-rose-500"
   480→                    onClick={() => removeItem(idx)}
   481→                    disabled={items.length === 1}
   482→                  >
   483→                    <X className="w-3.5 h-3.5" />
   484→                  </Button>
   485→                </div>
   486→              ))}
   487→            </div>
   488→          </div>
   489→
   490→          {/* Discount / tax / payment */}
   491→          <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   492→            <div className="space-y-1.5">
   493→              <Label className="text-xs">Discount (Rs)</Label>
   494→              <Input type="number" value={discount} onChange={(e) => setDiscount(Number(e.target.value))} className="h-8" />
   495→            </div>
   496→            <div className="space-y-1.5">
   497→              <Label className="text-xs">Tax Rate (%)</Label>
   498→              <Input type="number" value={taxRate} onChange={(e) => setTaxRate(Number(e.target.value))} className="h-8" />
   499→            </div>
   500→            <div className="space-y-1.5">
   501→              <Label className="text-xs">Payment Method</Label>
   502→              <Select value={paymentMethod} onValueChange={setPaymentMethod}>
   503→                <SelectTrigger className="h-8"><SelectValue /></SelectTrigger>
   504→                <SelectContent>
   505→                  {PAYMENT_METHODS.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   506→                </SelectContent>
   507→              </Select>
   508→            </div>
   509→            <div className="space-y-1.5">
   510→              <Label className="text-xs">Paid (Rs)</Label>
   511→              <Input type="number" value={paid} onChange={(e) => setPaid(Number(e.target.value))} className="h-8" />
   512→            </div>

   537→/* ---------- Collect Payment Dialog ---------- */
   538→
   539→function CollectPaymentDialog({
   540→  invoice, onOpenChange, onPaid,
   541→}: {
   542→  invoice: Invoice | null;
   543→  onOpenChange: (v: boolean) => void;
   544→  onPaid: () => void;
   545→}) {
   546→  const [amount, setAmount] = useState(0);
   547→  const [method, setMethod] = useState("Cash");
   548→  const [saving, setSaving] = useState(false);
   549→
   550→  // Reset amount when invoice changes
   551→  useEffect(() => {
   552→    if (invoice) {
   553→      setAmount(invoice.due || 0);
   554→      setMethod(invoice.paymentMethod || "Cash");
   555→    }
   556→  }, [invoice]);
   557→
   558→  if (!invoice) return null;
   559→
   560→  const newPaidTotal = (invoice.paid || 0) + (Number(amount) || 0);
   561→  const newDue = Math.max(0, invoice.total - newPaidTotal);
   562→  const newStatus = newDue <= 0 ? "paid" : newPaidTotal > 0 ? "partial" : "unpaid";
   563→
   564→  const submit = async (e: React.FormEvent) => {
   565→    e.preventDefault();
   566→    if (amount <= 0) { toast.error("Enter a valid amount"); return; }
   567→    setSaving(true);
   568→    try {
   569→      const res = await fetch(`/api/invoices/${invoice.id}`, {
   570→        method: "PATCH",
   571→        headers: { "Content-Type": "application/json" },
   572→        body: JSON.stringify({ paid: newPaidTotal, status: newStatus, paymentMethod: method }),
   573→      });
   574→      if (!res.ok) throw new Error("Failed to record payment");
   575→      toast.success(`Payment of ${formatRs(amount)} recorded for ${invoice.invoiceNo}`);
   576→      onPaid();
   577→    } catch {
   578→      toast.error("Failed to record payment");
   579→    } finally {
   580→      setSaving(false);
   581→    }
   582→  };
   583→
   584→  return (
   585→    <Dialog open={!!invoice} onOpenChange={onOpenChange}>
   586→      <DialogContent className="max-w-md">
   587→        <DialogHeader>
   588→          <DialogTitle className="flex items-center gap-2">
   589→            <CreditCard className="w-4 h-4 text-teal-600" /> Collect Payment
   590→          </DialogTitle>
   591→          <DialogDescription>
   592→            Invoice <span className="font-mono">{invoice.invoiceNo}</span> · {invoice.patient.name}
   593→          </DialogDescription>
   594→        </DialogHeader>
   595→        <form onSubmit={submit} className="space-y-3">
   596→          <div className="rounded-lg border bg-muted/30 p-3 text-sm space-y-1">
   597→            <div className="flex justify-between"><span className="text-muted-foreground">Invoice Total</span><span className="font-medium">{formatRs(invoice.total)}</span></div>
   598→            <div className="flex justify-between"><span className="text-muted-foreground">Already Paid</span><span className="text-emerald-600">{formatRs(invoice.paid)}</span></div>
   599→            <div className="flex justify-between font-semibold"><span>Outstanding Due</span><span className="text-rose-600">{formatRs(invoice.due)}</span></div>
   600→          </div>
   601→          <div className="grid grid-cols-2 gap-3">
   602→            <div className="space-y-1.5">
   603→              <Label>Amount (Rs) *</Label>
   604→              <Input type="number" value={amount} onChange={(e) => setAmount(Number(e.target.value))} min={0} max={invoice.due} />
   605→            </div>
   606→            <div className="space-y-1.5">
   607→              <Label>Method</Label>
   608→              <Select value={method} onValueChange={setMethod}>
   609→                <SelectTrigger><SelectValue /></SelectTrigger>
   610→                <SelectContent>
   611→                  {PAYMENT_METHODS.map((m) => <SelectItem key={m} value={m}>{m}</SelectItem>)}
   612→                </SelectContent>
   613→              </Select>
   614→            </div>
   615→          </div>
   616→          <div className="rounded-lg border bg-teal-50 dark:bg-teal-950/20 p-3 text-sm space-y-1">
   617→            <div className="flex justify-between"><span className="text-muted-foreground">New Paid Total</span><span className="text-emerald-600 font-medium">{formatRs(newPaidTotal)}</span></div>
   618→            <div className="flex justify-between"><span className="text-muted-foreground">Remaining Due</span><span className="font-medium">{formatRs(newDue)}</span></div>
   619→            <div className="flex justify-between"><span className="text-muted-foreground">New Status</span>
   620→              <Badge className={`text-[10px] ${statusColors[newStatus]}`}>{statusLabel(newStatus)}</Badge>
   621→            </div>
   622→          </div>
   623→          <DialogFooter>
   624→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   625→            <Button type="submit" disabled={saving} className="bg-teal-600 hover:bg-teal-700 text-white">
   626→              {saving ? "Recording…" : "Record Payment"}
   627→            </Button>
   628→          </DialogFooter>
   629→        </form>
   630→      </DialogContent>
   631→    </Dialog>
   632→  );
   633→}
