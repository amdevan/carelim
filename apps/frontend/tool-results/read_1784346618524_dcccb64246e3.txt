   190→export default function AdminPage() {
   191→  const [authed, setAuthed] = useState(false);
   192→  const [tab, setTab] = useState<ViewTab>("dashboard");
   193→  const [filter, setFilter] = useState<string>("");
   194→  const [sidebarCollapsed, setSidebarCollapsed] = useState(false);
   195→  const [mobileSidebarOpen, setMobileSidebarOpen] = useState(false);
   196→  const [expandedItems, setExpandedItems] = useState<Set<string>>(new Set(["Tenants"]));
   197→  const [search, setSearch] = useState("");
   198→  const { theme, setTheme } = useTheme();
   199→  const [mounted, setMounted] = useState(false);
   200→
   201→  // eslint-disable-next-line react-hooks/set-state-in-effect
   202→  useEffect(() => { if (!mounted) setMounted(true); }, [mounted]);
   203→
   204→  // Build flat list of all nav children for global search
   205→  const allNavChildren = useMemo(() => {
   206→    const list: { label: string; tab: ViewTab; action?: string; parent: string }[] = [];
   207→    NAV_GROUPS.forEach((g) => {
   208→      g.items.forEach((item) => {
   209→        if (item.tab) list.push({ label: item.label, tab: item.tab, parent: g.label });
   210→        (item.children || []).forEach((c) => list.push({ label: c.label, tab: c.tab, action: c.action, parent: item.label }));
   211→      });
   212→    });
   213→    return list;
   214→  }, []);
   215→
   216→  const searchResults = useMemo(() => {
   217→    if (!search) return [];
   218→    const q = search.toLowerCase();
   219→    return allNavChildren.filter((c) => c.label.toLowerCase().includes(q) || c.parent.toLowerCase().includes(q)).slice(0, 8);
   220→  }, [search, allNavChildren]);
   221→
   222→  const navigate = (newTab: ViewTab, action?: string) => {
   223→    setTab(newTab);
   224→    setFilter(action || "");
   225→    setMobileSidebarOpen(false);
   226→    setSearch("");
   227→  };
   228→
   229→  const toggleExpand = (label: string) => {
   230→    setExpandedItems((prev) => {
   231→      const next = new Set(prev);
   232→      if (next.has(label)) next.delete(label); else next.add(label);
   233→      return next;
   234→    });
   235→  };
   236→
   237→  if (!authed) return <SaasLogin onLogin={() => setAuthed(true)} />;
   238→
   239→  return (
   240→    <div className="flex min-h-screen bg-background">
   241→      {/* Desktop Sidebar */}
   242→      <motion.aside
   243→        animate={{ width: sidebarCollapsed ? 76 : 264 }}
   244→        transition={{ type: "spring", stiffness: 260, damping: 30 }}
   245→        className="hidden md:flex flex-col shrink-0 border-r border-sidebar-border bg-sidebar sticky top-0 h-screen z-30"
   246→      >
   247→        <SidebarBrand collapsed={sidebarCollapsed} />
   248→        <SidebarNav
   249→          collapsed={sidebarCollapsed}
   250→          expandedItems={expandedItems}
   251→          toggleExpand={toggleExpand}
   252→          activeTab={tab}
   253→          activeFilter={filter}
   254→          navigate={navigate}
   255→        />
   256→        <SidebarFooter collapsed={sidebarCollapsed} />
   257→      </motion.aside>
   258→
   259→      {/* Mobile Sidebar */}
   260→      <AnimatePresence>
   261→        {mobileSidebarOpen && (
   262→          <>
   263→            <motion.div
   264→              initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   265→              className="md:hidden fixed inset-0 bg-black/40 backdrop-blur-sm z-40"
   266→              onClick={() => setMobileSidebarOpen(false)}
   267→            />
   268→            <motion.aside
   269→              initial={{ x: -300 }} animate={{ x: 0 }} exit={{ x: -300 }}
   270→              transition={{ type: "spring", stiffness: 260, damping: 30 }}
   271→              className="md:hidden fixed left-0 top-0 bottom-0 w-72 bg-sidebar border-r border-sidebar-border z-50 flex flex-col"
   272→            >
   273→              <div className="flex items-center justify-between p-4 border-b border-sidebar-border">
   274→                <SidebarBrand collapsed={false} compact />
   275→                <Button variant="ghost" size="icon" className="h-8 w-8" onClick={() => setMobileSidebarOpen(false)}>
   276→                  <X className="w-4 h-4" />
   277→                </Button>
   278→              </div>
   279→              <SidebarNav
   280→                collapsed={false}
   281→                expandedItems={expandedItems}
   282→                toggleExpand={toggleExpand}
   283→                activeTab={tab}
   284→                activeFilter={filter}
   285→                navigate={navigate}
   286→              />
   287→              <SidebarFooter collapsed={false} />
   288→            </motion.aside>
   289→          </>
   290→        )}
   291→      </AnimatePresence>
   292→
   293→      {/* Main */}
   294→      <div className="flex-1 flex flex-col min-w-0">
   295→        <AdminHeader
   296→          tab={tab}
   297→          filter={filter}
   298→          theme={theme}
   299→          mounted={mounted}
   300→          setTheme={setTheme}
   301→          toggleSidebar={() => setSidebarCollapsed((v) => !v)}
   302→          openMobile={() => setMobileSidebarOpen(true)}
   303→          search={search}
   304→          setSearch={setSearch}
   305→          searchResults={searchResults}
   306→          navigate={navigate}
   307→          logout={() => { setAuthed(false); toast.success("Signed out"); }}
   308→        />
   309→
   310→        <main className="flex-1 p-4 sm:p-5 lg:p-6">
   311→          <AnimatePresence mode="wait">
   312→            <motion.div
   313→              key={tab + filter}
   314→              initial={{ opacity: 0, y: 6 }}
   315→              animate={{ opacity: 1, y: 0 }}
   316→              exit={{ opacity: 0 }}
   317→              transition={{ duration: 0.2 }}
   318→            >
   319→              <ViewRenderer tab={tab} filter={filter} />
   320→            </motion.div>
   321→          </AnimatePresence>
   322→        </main>
   323→      </div>
   324→    </div>
   325→  );
   326→}

   331→function ViewRenderer({ tab, filter }: { tab: ViewTab; filter: string }) {
   332→  switch (tab) {
   333→    case "dashboard": return <SaasDashboard />;
   334→    case "tenants": return <SaasTenants filter={filter} />;
   335→    case "subscriptions": return <SaasSubscriptions filter={filter} />;
   336→    case "billing": return <SaasBilling filter={filter} />;
   337→    case "users": return <SaasUsers />;
   338→    case "modules": return <SaasModules filter={filter} />;
   339→    case "addons": return <SaasAddOns />;
   340→    case "analytics": return <SaasAnalytics />;
   341→    case "support": return <SaasSupport filter={filter} />;
   342→    case "crm": return <SaasCRM filter={filter} />;
   343→    case "integrations": return <SaasIntegrations />;
   344→    case "security": return <SaasSecurity filter={filter} />;
   345→    case "settings": return <SaasSettings />;
   346→    default: return <SaasDashboard />;
   347→  }

   418→function SidebarItem({
   419→  item, collapsed, expanded, toggleExpand, activeTab, activeFilter, navigate,
   420→}: {
   421→  item: NavItem;
   422→  collapsed: boolean;
   423→  expanded: boolean;
   424→  toggleExpand: () => void;
   425→  activeTab: ViewTab;
   426→  activeFilter: string;
   427→  navigate: (tab: ViewTab, action?: string) => void;
   428→}) {
   429→  const Icon = item.icon;
   430→  const hasChildren = !!item.children?.length;
   431→  const isActive = item.tab === activeTab;
   432→  const childActive = hasChildren && (item.children || []).some((c) => c.tab === activeTab && (!c.action || c.action === activeFilter));
   433→
   434→  return (
   435→    <div>
   436→      <button
   437→        onClick={() => {
   438→          if (hasChildren) toggleExpand();
   439→          else if (item.tab) navigate(item.tab);
   440→        }}
   441→        title={collapsed ? item.label : undefined}
   442→        className={cn(
   443→          "group relative flex items-center gap-3 w-full rounded-lg px-2.5 py-2 text-sm font-medium transition-all duration-200",
   444→          (isActive || childActive)
   445→            ? "bg-gradient-to-r from-teal-600 to-emerald-600 text-white shadow-md shadow-teal-500/20"
   446→            : "text-sidebar-foreground/75 hover:bg-teal-50 dark:hover:bg-teal-950/30",
   447→          collapsed && "justify-center"
   448→        )}
   449→      >
   450→        <Icon className="w-[18px] h-[18px] shrink-0" />
   451→        <AnimatePresence>
   452→          {!collapsed && (
   453→            <motion.span
   454→              initial={{ opacity: 0, x: -6 }}
   455→              animate={{ opacity: 1, x: 0 }}
   456→              exit={{ opacity: 0, x: -6 }}
   457→              className="flex-1 text-left truncate"
   458→            >
   459→              {item.label}
   460→            </motion.span>
   461→          )}
   462→        </AnimatePresence>
   463→        {!collapsed && hasChildren && (
   464→          <ChevronDown className={cn("w-3.5 h-3.5 shrink-0 transition-transform", expanded && "rotate-180")} />
   465→        )}
   466→      </button>
   467→
   468→      <AnimatePresence>
   469→        {!collapsed && hasChildren && expanded && (
   470→          <motion.div
   471→            initial={{ opacity: 0, height: 0 }}
   472→            animate={{ opacity: 1, height: "auto" }}
   473→            exit={{ opacity: 0, height: 0 }}
   474→            transition={{ duration: 0.2 }}
   475→            className="overflow-hidden"
   476→          >
   477→            <div className="ml-4 pl-3 border-l border-sidebar-border space-y-0.5 mt-0.5">
   478→              {(item.children || []).map((c) => {
   479→                const active = c.tab === activeTab && (c.action || "") === activeFilter;
   480→                return (
   481→                  <button
   482→                    key={c.label}
   483→                    onClick={() => navigate(c.tab, c.action)}
   484→                    className={cn(
   485→                      "flex items-center gap-2 w-full rounded-md px-2.5 py-1.5 text-[13px] transition-all",
   486→                      active
   487→                        ? "bg-teal-50 dark:bg-teal-950/40 text-teal-700 dark:text-teal-300 font-medium"
   488→                        : "text-muted-foreground hover:bg-muted/60 hover:text-foreground"
   489→                    )}
   490→                  >
   491→                    <span className={cn("w-1 h-1 rounded-full", active ? "bg-teal-600" : "bg-muted-foreground/40")} />
   492→                    <span className="truncate">{c.label}</span>
   493→                  </button>
   494→                );
   495→              })}
   496→            </div>
   497→          </motion.div>
   498→        )}
   499→      </AnimatePresence>
   500→    </div>
   501→  );
   502→}
